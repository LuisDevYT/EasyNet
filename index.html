PlayerController.cs 
using UnityEngine;

public class PlayerController : DeterministicBehaviour
{
    public sfloat moveSpeed = (sfloat)5;

    public override void OnInput(PlayerInput input)
    {
        // This is the only thing you need.
        // Read input and move the player.

        sfloat x = input.GetSFloat("Horizontal");
        sfloat z = input.GetSFloat("Vertical");

        sfloat3 direction = new sfloat3(x, 0, z);

        transform.position += direction * moveSpeed * Deterministic.DeltaTime;
    }
}
Bomb.cs 
using UnityEngine;
using System.Collections;

public class Bomb : DeterministicBehaviour
{
    public GameObject explosionPrefab;

    public override void OnSpawn()
    {
        StartDeterministicCoroutine(ExplodeAfterDelay());
    }

    IEnumerator ExplodeAfterDelay()
    {
        yield return new WaitForDeterministicSeconds((sfloat)2);

        Deterministic.Spawn(
            explosionPrefab,
            transform.position,
            transform.rotation
        );

        Deterministic.Destroy(gameObject);
    }
}
